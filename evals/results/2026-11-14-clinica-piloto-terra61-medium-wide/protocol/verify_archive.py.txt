"""Confere o manifesto desta pasta e os resumos associados, sem acessar segredos."""

import hashlib
import json
from pathlib import Path

archive = Path(__file__).resolve().parents[1]
project = archive.parents[2]
manifest = json.loads((archive / "manifest.json").read_text(encoding="utf-8"))
expected = {row["path"] for row in manifest["files"]}
cache_names = {".ruff_cache", ".mypy_cache", ".pytest_cache", "__pycache__"}
actual = {path.relative_to(archive).as_posix() for path in archive.rglob("*") if path.is_file() and not cache_names.intersection(path.relative_to(archive).parts)}
assert actual == expected | {"manifest.json"}, "Arquivos ausentes ou fora do catálogo"
for row in manifest["files"]:
    data = (archive / row["path"]).read_bytes()
    assert len(data) == row["bytes"], row["path"]
    assert hashlib.sha256(data).hexdigest() == row["sha256"], row["path"]
for row in manifest.get("measurement_files", []):
    data = (project / row["path"]).read_bytes()
    assert hashlib.sha256(data).hexdigest() == row["sha256"], row["path"]
print(f"Integridade conferida: {len(expected)} arquivos e {len(manifest.get('measurement_files', []))} resumos.")
