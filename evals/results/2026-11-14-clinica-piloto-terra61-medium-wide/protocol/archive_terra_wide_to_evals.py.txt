def read_json(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def digest(data):
    return hashlib.sha256(data).hexdigest()


FORMS = set().union(*(secret_forms(p) for p in ATTEMPTS))
FORMS |= {base64.b64encode(s.encode()).decode() for s in FORMS}
COUNTS = Counter()
FILES = []
OMITTED = []
PUBLIC_MESSAGES = set()
for attempt in ATTEMPTS:
    coordinator = attempt.with_name(attempt.name + "-coordinator")
    for cli in coordinator.glob("codex-*.jsonl"):
        for line in cli.read_text(encoding="utf-8-sig").splitlines():

            item = event.get("item", {})
            if item.get("type") == "agent_message":
                PUBLIC_MESSAGES.add(item.get("text", ""))

JWT = re.compile(r"\beyJ[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}\b")
AUTH = re.compile(r"(?i)(\b(?:Bearer|Basic)\s+)([A-Za-z0-9_+/=.~-]{12,})")
HEADER = re.compile(r"(?im)^(\s*)(Cookie|Set-Cookie|X-XSRF-TOKEN|X-CSRF-TOKEN|X-ZAP-API-Key)(\s*:\s*)([^\r\n]+)")
SECRET_KEYS = {"authorization", "cookie", "set-cookie", "x-xsrf-token", "x-csrf-token", "x-zap-api-key", "access_token", "refresh_token", "id_token", "api_key", "apikey", "password", "senha"}


def session_header(match):
    prefix = "".join(match.group(i) for i in (1, 2, 3))
    name = match.group(2).lower()
    if name in {"cookie", "set-cookie"}:
        parts = match.group(4).split(";")
        count = len(parts) if name == "cookie" else 1
        for i in range(count):
            key, separator, value = parts[i].partition("=")
            if separator:
                parts[i] = key + "=[REDACTED:cookie-value]"
            elif parts[i].strip():
                parts[i] = "[REDACTED:cookie-value]"
        return prefix + ";".join(parts)
    return prefix + "[REDACTED:session-header]"


def sanitize_text(text):
    for form in sorted(FORMS, key=len, reverse=True):
        n = text.count(form)
        if n:
            COUNTS["known_round_secret"] += n
            text = text.replace(form, "[REDACTED:round-secret]")
    text, n = JWT.subn("[REDACTED:jwt]", text)
    COUNTS["jwt"] += n
    text, n = AUTH.subn(r"\1[REDACTED:authorization]", text)
    COUNTS["authorization_value"] += n
    text, n = HEADER.subn(session_header, text)
    COUNTS["session_header"] += n
    return text


def sanitize(value, key=""):
    if isinstance(value, dict):
        return {k: sanitize(v, k) for k, v in value.items()}
    if isinstance(value, list):
        return [sanitize(v, key) for v in value]
    if isinstance(value, str):
        if key.lower() in SECRET_KEYS and value and not value.startswith(("env:", "[REDACTED", "***")):
            COUNTS["credential_or_session_field"] += 1
            return "[REDACTED:credential-or-session-field]"
        return sanitize_text(value)
    return value


def safe_event(event):
    item = event.get("item", {})
    if item and item.get("type") not in {"agent_message", "command_execution"}:
        COUNTS["non_public_event_omitted"] += 1
        return None
    if event.get("type") not in {"thread.started", "turn.started", "turn.completed", "turn.failed", "item.started", "item.completed", "item.updated", "error"}:
        COUNTS["unknown_event_omitted"] += 1
        return None
    return event


def public_transcript(row):
    msg = row.get("message", {})
    content = msg.get("content")
    if row.get("type") == "assistant" and isinstance(content, str) and content not in PUBLIC_MESSAGES:
        # A exportação original não guardava o canal; a CLI pública confirma o texto permitido.
        msg["content"] = []
        COUNTS["unverified_assistant_text_omitted"] += 1
    return row


def store(relative, raw, source=None, transformations=(), json_value=None, formatted_text=None):
    target = DEST / relative
    if target.exists():
        raise RuntimeError(f"Destino já existe: {target.relative_to(PROJECT)}")
    target.parent.mkdir(parents=True, exist_ok=True)
    before = COUNTS.copy()
    if formatted_text is not None:
        output = sanitize_text(formatted_text)
    elif json_value is not None:
        output = json.dumps(sanitize(json_value), ensure_ascii=False, indent=2) + "\n"
    else:
        output = sanitize_text(raw.decode("utf-8-sig"))
    for attempt in ATTEMPTS:
        ensure_no_secrets(attempt, {relative: output})
    encoded = output.encode("utf-8")
    target.write_bytes(encoded)
    changes = {k: v - before[k] for k, v in COUNTS.items() if v > before[k]}
    FILES.append({"path": relative, "bytes": len(encoded), "sha256": digest(encoded), "source": str(source) if source else "generated:archive_terra_wide_to_evals.py", "source_sha256": digest(raw) if source else None, "transformations": list(transformations), "redactions": changes})


def generated(relative, text):
    store(relative, text.encode("utf-8"))


def generated_json(relative, data):
    store(relative, b"", json_value=data)


def copy_file(source, relative, mode=None):
    if not source.is_file():
        raise FileNotFoundError(str(source))
    raw = source.read_bytes()
    transforms = ["UTF-8; credential/session redaction when matched"]
    if relative.endswith((".json", ".jsonl")):
        if relative.endswith(".json"):
            store(relative, raw, source, transforms, read_json(source))
            return
        rows = []
        before = COUNTS.copy()
        for line in raw.decode("utf-8-sig").splitlines():
            if not line.strip():
                continue
            row = json.loads(line)
            if mode == "cli":
                row = safe_event(row)
            elif mode == "timed":
                event = safe_event(row["event"])
                row = {**row, "event": event} if event else None
            elif mode == "transcript":
                row = public_transcript(row)
            elif mode == "resources":
                containers = row.get("containers", [])
                own = [c for c in containers if c.get("Name", "").startswith("hdast-terra-wide")]
                row = {**row, "containers": own, "other_container_count": len(containers) - len(own)}
            if row is not None:
                rows.append(sanitize(row))
        transforms.append({"cli": "public CLI events only", "timed": "timestamped public CLI events only", "transcript": "tools, results, usage and assistant text independently confirmed in public CLI", "resources": "own containers only; retain count of other containers"}.get(mode, "JSONL normalization"))
        out = "".join(json.dumps(row, ensure_ascii=False) + "\n" for row in rows)
        store(relative, raw, source, transforms + ["JSON serialization"], formatted_text=out)
        FILES[-1]["redactions"] = {k: v - before[k] for k, v in COUNTS.items() if v > before[k]}
    else:
        store(relative, raw, source, transforms)


def archive_name(name):
    return name + ".txt" if name.endswith((".md", ".py", "pyproject.toml")) else name


def copy_tree(source, relative):
    for path in sorted(source.rglob("*")):
        if not path.is_file():
            continue
        parts = path.relative_to(source).parts
        if any(x in {"__pycache__", "sealed", "state", "approvals", ".venv", ".git"} for x in parts):
            OMITTED.append({"source": str(path), "reason": "excluded private/state/environment directory"})
            continue
        mapped = ["executions" if x == "runs" else x for x in parts]
        mapped[-1] = archive_name(mapped[-1])
        mode = "transcript" if path.name == "transcript.jsonl" else "cli" if path.name.startswith("codex-") and path.suffix == ".jsonl" else "timed" if path.name == "timed-events.jsonl" else None
        copy_file(path, (Path(relative) / Path(*mapped)).as_posix(), mode)


def csv_text(rows):
    output = io.StringIO(newline="")
    writer = csv.DictWriter(output, fieldnames=list(rows[0]))
    writer.writeheader()
    writer.writerows(rows)
    return output.getvalue()
