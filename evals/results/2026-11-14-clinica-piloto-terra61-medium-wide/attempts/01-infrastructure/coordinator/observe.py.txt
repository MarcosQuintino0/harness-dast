"""Lê somente métricas e estado público da rodada em curso."""

import argparse
import json
import re
import subprocess
import time
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path(__file__).parent
WORKSPACE = ROOT.with_name(ROOT.name.removesuffix("-coordinator")) / "workspace"


def documents(path):
    if not path.exists():
        return []
    rows = []
    for line in path.read_text(encoding="utf-8", errors="replace").splitlines():
        try:
            rows.append(json.loads(line))
        except json.JSONDecodeError:
            continue
    return rows


def rollout():
    marker = ROOT / "thread-id.txt"
    if not marker.exists():
        return None
    thread = marker.read_text(encoding="utf-8").strip()
    candidates = list((Path.home() / ".codex/sessions").glob(f"2026/11/*/*{thread}.jsonl"))
    return candidates[0] if candidates else None


def status():
    events = documents(ROOT / "timed-events.jsonl")
    commands = []
    messages = []
    for row in events:
        event = row["event"]
        item = event.get("item", {})
        if event.get("type") == "item.started" and item.get("type") == "command_execution":
            found = re.findall(r"hdast\s+(?:--json\s+)?([a-z-]+)(?:\s+([a-z-]+))?", item.get("command", ""))
            commands.append({"at": row["timestamp"], "hdast": [" ".join(parts).strip() for parts in found],
                             "kind": "hdast" if found else "local files"})
        if event.get("type") == "item.completed" and item.get("type") == "agent_message":
            messages.append(item.get("text", "")[:600])
    result = {"events": len(events), "recent_commands": commands[-4:], "recent_messages": messages[-1:]}
    native = rollout()
    if native:
        for row in documents(native):
            payload = row.get("payload", {})
            if row.get("type") == "turn_context":
                result["model"] = payload.get("model")
                result["effort"] = payload.get("effort", payload.get("reasoning_effort"))
            if row.get("type") == "event_msg" and payload.get("type") == "token_count":
                result["tokens"] = (payload.get("info") or {}).get("total_token_usage")
    result["profiles"] = [str(path.relative_to(WORKSPACE)) for path in (WORKSPACE / "products").glob("*/target.yml")]
    result["runs"] = [str(path.relative_to(WORKSPACE)) for path in (WORKSPACE / "runs").glob("*/run.json")]
    print(json.dumps(result, ensure_ascii=False))


def resources():
    while not (ROOT / "STOP-RESOURCE-MONITOR").exists():
        started = time.monotonic()
        command = ["docker", "stats", "--no-stream", "--format", "{{json .}}"]
        result = subprocess.run(command, capture_output=True, text=True, encoding="utf-8", timeout=20)
        rows = []
        for line in result.stdout.splitlines():
            if line.strip():
                rows.append(json.loads(line))
        sample = {"timestamp": datetime.now(UTC).isoformat(), "containers": rows,
                  "sampling_seconds": round(time.monotonic() - started, 3)}
        with (ROOT / "resources.jsonl").open("a", encoding="utf-8") as output:
            output.write(json.dumps(sample) + "\n")
        time.sleep(30)


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("action", choices=["status", "resources"])
    if parser.parse_args().action == "resources":
        resources()
    else:
        status()
