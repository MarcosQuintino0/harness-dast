"""Diagnóstico da API de controle do ZAP, sem chamadas à API do produto."""

import json
import re
import sys

import httpx

from runner import WORKSPACE, agent_env

for name, value in agent_env().items():
    import os
    os.environ[name] = value
sys.path.insert(0, str(WORKSPACE / "src"))
from harness_dast.cli.stack_composition import load_effective_settings

settings = load_effective_settings()
key = settings.zap_api_key.get_secret_value() if settings.zap_api_key else ""
with httpx.Client(trust_env=False, timeout=15, headers={"X-ZAP-API-KEY": key}) as client:
    response = client.get(settings.zap_base_url + "/JSON/core/view/version/")
try:
    body = response.json()
except ValueError:
    body = None
version = body.get("version") if isinstance(body, dict) else None
summary = {"base_url": settings.zap_base_url, "key_configured": bool(key),
           "status": response.status_code, "content_type": response.headers.get("content-type"),
           "response_bytes": len(response.content), "json_keys": list(body) if isinstance(body, dict) else None,
           "version": version if isinstance(version, str) and re.fullmatch(r"[0-9.]+", version) else None}
if response.status_code == 403 and len(response.content) < 120:
    summary["short_control_api_error"] = response.text.replace(key, "[redacted]") if key else response.text
print(json.dumps(summary))
