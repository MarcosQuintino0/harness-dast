def read(path):
    return json.loads(path.read_text(encoding="utf-8"))


def number(value):
    return f"{value:,}".replace(",", ".")


def duration(seconds):
    minutes, remaining = divmod(seconds, 60)
    return f"{int(minutes)} min {remaining:05.2f} s"


def stage_rows(metrics):
    labels = {"leitura": "Leitura e retomada", "cadastro": "Cadastro e perfil",
              "alcance": "Alcançabilidade", "varredura": "Pilha, preflight e execução",
              "triagem": "Triagem e relatório", "outro": "Outros"}
    rows = []
    for key, label in labels.items():
        stage = metrics["stages"][key]
        if not stage["tool_calls"] and stage["tokens"] is None:
            continue
        tokens = stage["tokens"] or {}
        rows.append(f"| {label} | {stage['seconds']:.3f} | {number(tokens.get('input', 0))} | "
                    f"{number(tokens.get('cache_read', 0))} | {number(tokens.get('output', 0))} | {stage['tool_calls']} |")
    return "\n".join(rows)
