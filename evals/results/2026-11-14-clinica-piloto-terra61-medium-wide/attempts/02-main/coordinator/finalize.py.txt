"""Consolida medidas verificáveis da rodada e da continuação assistida."""

import json
import re
import shutil
import subprocess
import sys
from collections import Counter
from datetime import date
from pathlib import Path

SOURCE = Path("C:/harness-dast-wt/v17-baseline-terra-medium")
ROOT = Path(__file__).parent
ROUND = ROOT.with_name(ROOT.name.removesuffix("-coordinator"))
RUN_ID = "2026-11-15T01-31-02Z-clinica-piloto-3c63"
RUN = ROUND / "workspace/runs" / RUN_ID
sys.path.insert(0, str(SOURCE))

from evals.runner.measurements import ensure_no_secrets, save_measurement
from evals.runner.metrics import measure_agent


def read(path):
    return json.loads(path.read_text(encoding="utf-8"))


def write(path, value):
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")


def memory_mib(text):
    match = re.fullmatch(r"\s*([\d.]+)\s*(B|KiB|MiB|GiB)\s*", text)
    if not match:
        raise ValueError("Unidade de memória inesperada")
    return float(match[1]) * {"B": 1 / 1048576, "KiB": 1 / 1024, "MiB": 1, "GiB": 1024}[match[2]]


telemetry = read(ROOT / "telemetry.json")
autonomous = read(ROOT / "autonomous-boundary/agent-metrics.json")
autonomous_tokens = read(ROOT / "autonomous-boundary/telemetry.json")["totals"]
complete = read(ROUND / "agent-metrics.json")
last_turn = list(telemetry["requests"].values())[-1]["turn_id"]
segment = ROOT / "assisted-segment"
segment.mkdir(exist_ok=True)
with (segment / "transcript.jsonl").open("w", encoding="utf-8") as output:
    for line in (ROUND / "transcript.jsonl").read_text(encoding="utf-8").splitlines():
        if json.loads(line).get("turn_id") == last_turn:
            output.write(line + "\n")
metrics_folder = segment / "trace/runs" / RUN_ID
metrics_folder.mkdir(parents=True, exist_ok=True)
shutil.copy2(RUN / "metrics.json", metrics_folder / "metrics.json")
assisted_model = measure_agent(segment, "terra61-medium-wide-assistido-segmento")
assisted = assisted_model.model_dump(mode="json")
write(ROOT / "assisted-segment-metrics.json", assisted)
assisted_tokens = {key: value - autonomous_tokens.get(key, 0)
                   for key, value in telemetry["totals"].items()}
normalized_tokens = assisted["totals"]["tokens"]
if sum(normalized_tokens.values()) != assisted_tokens["total_tokens"]:
    raise SystemExit("Tokens da continuação não conferem com a diferença nativa")

duration = round(complete["totals"]["duration_seconds"])
subprocess.run([sys.executable, "-m", "evals.runner", "score", "--out", str(ROUND),
                "--tokens", str(telemetry["totals"]["total_tokens"]),
                "--duration-seconds", str(duration)], cwd=SOURCE, check=True)
subprocess.run([sys.executable, "-m", "evals.runner", "classify", "--out", str(ROUND)],
               cwd=SOURCE, check=True)

run = read(RUN / "run.json")
findings = read(RUN / "findings.json")
metrics = read(RUN / "metrics.json")
scan = read(RUN / "scan.json")
authz = read(RUN / "authz.json")
triage = read(RUN / "triage.json")
score = read(ROUND / "score.json")

resources = []
for line in (ROOT / "resources.jsonl").read_text(encoding="utf-8").splitlines():
    sample = json.loads(line)
    if scan["started_at"] <= sample["timestamp"] <= scan["finished_at"]:
        resources.append(sample)
container_peaks = {}
combined_peaks = []
for sample in resources:
    combined = 0.0
    for container in sample["containers"]:
        name = container.get("Name", "")
        if not name.startswith("hdast-terra-wide"):
            continue
        used, limit = container["MemUsage"].split(" / ")
        used_mib = memory_mib(used)
        cpu = float(container["CPUPerc"].removesuffix("%"))
        combined += used_mib
        peaks = container_peaks.setdefault(name, {"memory_mib": 0, "cpu_percent": 0,
                                                  "limit_mib": memory_mib(limit)})
        peaks["memory_mib"] = max(peaks["memory_mib"], used_mib)
        peaks["cpu_percent"] = max(peaks["cpu_percent"], cpu)
    combined_peaks.append(combined)

probe_runs = []
for path in sorted((ROUND / "workspace/runs").glob("*/probe.json")):
    probe = read(path)
    if not probe.get("dry_run"):
        probe_runs.append({"run_id": path.parent.name, "started_at": probe["started_at"],
                           "finished_at": probe["finished_at"], "summary": probe["summary"]})

rules = sorted(metrics["rules"]["totals"], key=lambda row: row.get("time_ms") or 0, reverse=True)
operations = sorted(metrics["operations"], key=lambda row: row.get("duration_seconds") or 0, reverse=True)
suspicions = sum(bool(annotation.get("suspected_false_positive")) for annotation in triage["annotations"])
summary = {
    "metadata": read(ROOT / "metadata.json"), "mode_revealed_after_agent_finished": score["mode"],
    "autonomous_boundary": {"metrics": autonomous, "tokens_native": autonomous_tokens,
                             "grade": read(ROOT / "autonomous-boundary/score.json")["grade"],
                             "result": "interrompida pela skill antes da varredura"},
    "assisted_segment": {"metrics": assisted, "tokens_native": assisted_tokens},
    "whole_second_attempt": {"metrics": complete, "tokens_native": telemetry["totals"],

                                  - autonomous["totals"]["duration_seconds"]

    "initial_infrastructure_attempt": read(ROOT.with_name(ROOT.name.replace("wide-02", "wide-01")) / "telemetry.json")["totals"],
    "pipeline": run, "probe_history": probe_runs, "coverage": findings["coverage"],
    "findings_summary": findings["summary"], "triage_annotations": len(triage["annotations"]),
    "suspected_false_positive_annotations": suspicions,
    "authz": {"requests_sent": authz["requests_sent"], "checks": authz["checks"],
               "findings": len(authz["findings"]), "not_tested_entries": len(authz["not_tested"])},
    "traffic": scan["traffic"], "traffic_gaps": metrics["traffic"]["gaps"],
    "rules_top_by_accumulated_time": rules[:10],
    "operations_top_by_duration": [{key: operation.get(key) for key in
        ("operation_id", "duration_seconds", "requests", "outcome")} for operation in operations[:10]],
    "resources": {"scan_samples": len(resources), "interval_seconds": 30,
                   "combined_sampled_peak_memory_mib": round(max(combined_peaks, default=0), 2),
                   "containers": container_peaks},
    "score": score,
    "limits": ["Uma amostra, com continuação assistida e cobertura parcial das operações.",
               "Outra sessão usava o mesmo Docker durante a medição.",
               "As categorias de tempo do agente seguem as regras do avaliador e diferem das etapas da CLI.",
               "A contagem textual de comandos hdast pode incluir exemplos escritos em diários PowerShell.",
               "Tokens nativos foram reconciliados por response_id; cache integra input e raciocínio integra output.",
               "Os picos de recursos são amostrados, com intervalo de 30 s mais o tempo de coleta.",
               "As verificações de segredos do avaliador procuram os valores sorteados da rodada.",
               "O efeito isolado das novas diretrizes não foi medido por um experimento A/B."],
}
serialized = json.dumps(summary, indent=2, ensure_ascii=False) + "\n"
ensure_no_secrets(ROUND, {"resumo-medicoes.json": serialized})
(ROUND / "resumo-medicoes.json").write_text(serialized, encoding="utf-8")
saved = save_measurement(ROUND, measure_agent(ROUND, "terra61-medium-wide-assistido"),
                         "terra61-medium-wide-assistido", date(2026, 11, 14),
                         root=ROUND.parent / "medicoes")
experiment = {"source_commit": summary["metadata"]["source_commit"], "model": "gpt-6.1-terra",
              "effort": "medium", "path_mode": "wide", "assisted": True,
              "operations_scanned": findings["coverage"]["scans_run"],
              "agent_active_seconds": round(autonomous["totals"]["duration_seconds"]
                                            + assisted["totals"]["duration_seconds"], 3),
              "concurrent_other_lab": True, "summary_file": str(ROUND / "resumo-medicoes.json")}
write(saved / "experiment.json", experiment)
print(json.dumps({"summary_file": str(ROUND / "resumo-medicoes.json"), "saved": str(saved),
                  "assisted_seconds": assisted["totals"]["duration_seconds"],
                  "pause_seconds": summary["whole_second_attempt"]["pause_seconds"],
                  "resources": summary["resources"], "grade": score["grade"]}, ensure_ascii=False))
