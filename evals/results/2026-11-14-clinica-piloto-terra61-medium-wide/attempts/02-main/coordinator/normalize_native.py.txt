"""Exporta chamadas públicas e uso nativo de tokens, incluindo esperas assíncronas."""

import ast
import json
import re
import sys
from pathlib import Path

ROOT = Path(sys.argv[1]) if len(sys.argv) > 1 else Path(__file__).parent
ROUND = ROOT.with_name(ROOT.name.removesuffix("-coordinator"))
thread = (ROOT / "thread-id.txt").read_text(encoding="utf-8").strip()
rollout = next((Path.home() / ".codex/sessions").glob(f"2026/11/*/*{thread}.jsonl"))
rows = [json.loads(line) for line in rollout.read_text(encoding="utf-8").splitlines()]
commands = re.compile(r'''(?:\bcmd|"cmd")\s*:\s*("(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')''')
normalized = []
pending = []
requests = {}
model = "gpt-6.1-terra"
current_turn = None


def append(role, stamp, content, **extra):
    identifier = f"native-{len(normalized)}"
    message = {"id": identifier, "content": content, **extra}
    if role == "assistant":
        message["model"] = model
    row = {"type": role, "uuid": identifier, "timestamp": stamp, "turn_id": current_turn,
           "message": message}
    normalized.append(row)
    return row


def tool_input(payload):
    source = payload.get("input", "")
    if "tools.exec_command(" in source:
        literals = commands.findall(source)
        decoded = []
        for literal in literals:
            try:
                decoded.append(json.loads(literal) if literal.startswith('"') else ast.literal_eval(literal))
            except (ValueError, SyntaxError):
                raise SystemExit("Argumento de shell não reconhecido; conferir a exportação")
        if not decoded:
            raise SystemExit("Comando de shell dinâmico sem argumento literal; conferir a exportação")
        return "PowerShell", {"command": "\n".join(decoded)}
    if "tools.write_stdin(" in source:
        return "Wait", {"transport": "write_stdin"}
    return payload.get("name", "LocalTool"), {"source": source}


for row in rows:
    stamp = row["timestamp"]
    payload = row.get("payload", {})
    kind = payload.get("type")
    if row.get("type") == "turn_context":
        current_turn = payload.get("turn_id")
    if row.get("type") == "event_msg" and kind == "task_started":
        current_turn = payload.get("turn_id")
        append("user", stamp, "Pedido do dono do produto no EVAL-BRIEFING.md.")
    if row.get("type") == "response_item" and kind == "custom_tool_call":
        name, arguments = tool_input(payload)
        entry = append("assistant", stamp, [{"type": "tool_use", "id": payload["call_id"],
                                             "name": name, "input": arguments}])
        pending.append(entry)
    elif row.get("type") == "response_item" and kind == "custom_tool_call_output":
        output = payload.get("output", "")
        text = output if isinstance(output, str) else json.dumps(output, ensure_ascii=False)
        code = re.search(r'"exit_code"\s*:\s*(-?\d+)', text)
        append("user", stamp, [{"type": "tool_result", "tool_use_id": payload["call_id"],
                                "content": text, "is_error": bool(code and int(code[1]) != 0)}])
    elif row.get("type") == "response_item" and kind == "message" and payload.get("role") == "assistant":
        content = "\n".join(item.get("text", "") for item in payload.get("content", [])
                            if item.get("type") in {"output_text", "text"})
        if content:
            pending.append(append("assistant", stamp, content))
    elif row.get("type") == "token_usage_record":
        identifier = payload["response_id"]
        if identifier in requests:
            continue
        usage = payload["usage"]
        if not pending:
            pending.append(append("assistant", stamp, []))
        for entry in pending:
            entry["message"]["id"] = identifier
        pending[0]["message"]["usage"] = {
            "input_tokens": usage["input_tokens"] - usage.get("cached_input_tokens", 0),
            "output_tokens": usage["output_tokens"],
            "cache_read_input_tokens": usage.get("cached_input_tokens", 0),
            "cache_creation_input_tokens": usage.get("cache_write_input_tokens", 0),
        }
        requests[identifier] = {"timestamp": stamp, "turn_id": payload.get("turn_id"), "usage": usage,
                                "thread_token_usage": payload["thread_token_usage"]}
        pending = []

totals = {key: sum(record["usage"].get(key, 0) for record in requests.values())
          for key in ("input_tokens", "cached_input_tokens", "cache_write_input_tokens", "output_tokens",
                      "reasoning_output_tokens", "total_tokens")}
latest = list(requests.values())[-1]["thread_token_usage"]
if totals["total_tokens"] != latest["total_tokens"]:
    raise SystemExit("Total nativo diverge da soma por resposta")
with (ROUND / "transcript.jsonl").open("w", encoding="utf-8") as output:
    for row in normalized:
        output.write(json.dumps(row, ensure_ascii=False) + "\n")
telemetry = {"thread_id": thread, "model": model, "effort": "medium", "native_rollout": str(rollout),
             "api_responses": len(requests), "totals": totals, "native_totals_reconciled": True,
             "requests": requests, "normalization": "native public tool calls, results and assistant messages",
             "token_semantics": "input nativo inclui cache; reasoning integra output; exportação separa input sem cache",
             "stage_attribution": "regra existente do avaliador aplicada aos comandos de cada resposta do modelo"}
(ROOT / "telemetry.json").write_text(json.dumps(telemetry, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
print(json.dumps({"events": len(normalized), "api_responses": len(requests), "tokens": totals, "reconciled": True}))
