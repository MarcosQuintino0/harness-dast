"""Adapta eventos públicos do Codex ao formato de medição existente do laboratório.

Os arquivos nativos são preservados. A atribuição de tokens às etapas segue os
comandos relacionados temporalmente a cada resposta do modelo.
"""

import json
import sys
from datetime import datetime
from pathlib import Path

ROOT = Path(sys.argv[1]) if len(sys.argv) > 1 else Path(__file__).parent
ROUND = ROOT.with_name(ROOT.name.removesuffix("-coordinator"))


def read_jsonl(path):
    documents = []
    for line in path.read_text(encoding="utf-8", errors="replace").splitlines():
        try:
            documents.append(json.loads(line))
        except json.JSONDecodeError:
            continue
    return documents


def seconds(stamp):
    return datetime.fromisoformat(stamp.replace("Z", "+00:00")).timestamp()


events = read_jsonl(ROOT / "timed-events.jsonl")
thread = (ROOT / "thread-id.txt").read_text(encoding="utf-8").strip()
rollouts = list((Path.home() / ".codex/sessions").glob(f"2026/11/*/*{thread}.jsonl"))
if len(rollouts) != 1:
    raise SystemExit("Rollout exclusivo da rodada não encontrado")
native = read_jsonl(rollouts[0])
normalized = []
tool_messages = []


def append_message(role, identifier, timestamp, content, **extra):
    message = {"id": identifier, "content": content, **extra}
    if role == "assistant":
        message["model"] = "gpt-6.1-terra"
    event = {"type": role, "uuid": identifier, "timestamp": timestamp, "message": message}
    normalized.append(event)
    return event


for row in events:
    event = row["event"]
    item = event.get("item", {})
    identifier = "codex-" + row["invocation"] + "-" + item.get("id", "event")
    if item.get("type") == "command_execution":
        if event.get("type") == "item.started":
            record = append_message("assistant", identifier, row["timestamp"], [{
                "type": "tool_use", "id": identifier, "name": "PowerShell",
                "input": {"command": item.get("command", "")},
            }])
            tool_messages.append(record)
        elif event.get("type") == "item.completed":
            append_message("user", identifier + "-result", row["timestamp"], [{
                "type": "tool_result", "tool_use_id": identifier,
                "content": item.get("aggregated_output", ""),
                "is_error": item.get("exit_code", 0) not in (None, 0),
            }])
    elif item.get("type") == "agent_message" and event.get("type") == "item.completed":
        append_message("assistant", identifier, row["timestamp"], item.get("text", ""))

requests = {}
for row in native:
    payload = row.get("payload", {})
    if row.get("type") == "token_usage_record":
        requests[payload["response_id"]] = {"timestamp": row["timestamp"], **payload}

mapped = []
for response_id, record in requests.items():
    raw = record["usage"]
    usage = {
        "input_tokens": raw["input_tokens"] - raw.get("cached_input_tokens", 0),
        "output_tokens": raw["output_tokens"],
        "cache_read_input_tokens": raw.get("cached_input_tokens", 0),
        "cache_creation_input_tokens": raw.get("cache_write_input_tokens", 0),
    }
    closest = min(tool_messages,
                  key=lambda item: abs(seconds(item["timestamp"]) - seconds(record["timestamp"])),
                  default=None)
    difference = abs(seconds(closest["timestamp"]) - seconds(record["timestamp"])) if closest else None
    if closest and difference <= 3:
        target = closest["message"]
        totals = target.setdefault("usage", {})
        for name, value in usage.items():
            totals[name] = totals.get(name, 0) + value
        assigned = target["id"]
    else:
        append_message("assistant", response_id, record["timestamp"], [], usage=usage)
        assigned = response_id
    mapped.append({"response_id": response_id, "timestamp": record["timestamp"],
                   "usage": raw, "message_id": assigned,
                   "nearest_tool_seconds": difference if closest and difference <= 3 else None})

starts = [row["timestamp"] for row in native
          if row.get("type") == "event_msg" and row.get("payload", {}).get("type") == "task_started"]
if starts:
    append_message("user", "task-start", min(starts), "Pedido do dono do produto no EVAL-BRIEFING.md.")
normalized.sort(key=lambda event: seconds(event["timestamp"]))
totals = {name: sum(record["usage"].get(name, 0) for record in requests.values())
          for name in ("input_tokens", "cached_input_tokens", "cache_write_input_tokens",
                       "output_tokens", "reasoning_output_tokens", "total_tokens")}
latest = list(requests.values())[-1]["thread_token_usage"] if requests else {}
if totals.get("total_tokens") != latest.get("total_tokens"):
    raise SystemExit("Soma por resposta difere do total nativo; conferir antes de medir")
with (ROUND / "transcript.jsonl").open("w", encoding="utf-8") as output:
    for event in normalized:
        output.write(json.dumps(event, ensure_ascii=False) + "\n")
telemetry = {
    "thread_id": thread, "model": "gpt-6.1-terra", "effort": "medium",
    "native_rollout": str(rollouts[0]), "api_responses": len(requests),
    "totals": totals, "native_totals_reconciled": True, "requests": mapped,
    "stage_attribution": "comando iniciado mais próximo em até 3 s; demais respostas herdam a etapa corrente",
    "token_semantics": "input_tokens nativo inclui cached_input_tokens; reasoning_output_tokens integra output_tokens",
    "normalized_semantics": "input_tokens exclui cache; cache_read_input_tokens contém a parcela em cache",
}
(ROOT / "telemetry.json").write_text(json.dumps(telemetry, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
print(json.dumps({"normalized_events": len(normalized), "api_responses": len(requests),
                  "tokens": totals, "reconciled": True}))
